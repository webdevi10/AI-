
import heapq

initial = (
    (1, 2, 3),
    (0, 4, 6),
    (7, 5, 8)
)

goal = (
    (1, 2, 3),
    (4, 5, 6),
    (7, 8, 0)
)


def heuristic(state):
    distance = 0

    for i in range(3):
        for j in range(3):
            value = state[i][j]

            if value != 0:
                x = (value - 1) // 3
                y = (value - 1) % 3

                distance += abs(i - x) + abs(j - y)

    return distance


def get_neighbors(state):
    board = [list(row) for row in state]

    for i in range(3):
        for j in range(3):
            if board[i][j] == 0:
                x, y = i, j

    moves = [(-1, 0), (1, 0), (0, -1), (0, 1)]
    neighbors = []

    for dx, dy in moves:
        nx, ny = x + dx, y + dy

        if 0 <= nx < 3 and 0 <= ny < 3:
            new_board = [row[:] for row in board]

            new_board[x][y], new_board[nx][ny] = (
                new_board[nx][ny], new_board[x][y]
            )

            neighbors.append(
                tuple(tuple(row) for row in new_board)
            )

    return neighbors


def astar(initial, goal):
    queue = [(heuristic(initial), 0, initial, [initial])]
    visited = {}

    while queue:
        f, g, state, path = heapq.heappop(queue)

        if state == goal:
            print("Solution Path:")

            for step, board in enumerate(path):
                print("Step", step)
                for row in board:
                    print(row)
                print()

            print("Total Moves =", g)
            return

        if state in visited and visited[state] <= g:
            continue

        visited[state] = g

        for neighbor in get_neighbors(state):
            new_g = g + 1

            if neighbor not in visited or new_g < visited[neighbor]:
                new_f = new_g + heuristic(neighbor)
                heapq.heappush(
                    queue,
                    (new_f, new_g, neighbor, path + [neighbor])
                )

    print("No solution exists")


astar(initial, goal)
